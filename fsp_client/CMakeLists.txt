project(fsp_client CXX)

# ---------------------------------------------------------------------------
# IDL header generation
#
# IFSPApp/CFSPApp describe the FSP TA's own interface and CBO UID.
# IAppLoader/CAppLoader/IAppController are copied in from
# libmink/SDK/QTVM/idl (which has no CMake build of its own) and are used to
# push the fspapp ELF image to QTEE via IAppLoader_loadFromBuffer().
# ---------------------------------------------------------------------------

set(FSP_IDLS
	idl/IFSPApp
	idl/CFSPApp
	idl/IAppController
	idl/IAppLoader
	idl/CAppLoader
)

set(IDL_OUT_LIST "")

foreach(IDL ${FSP_IDLS})
	set(IDL_OUT   ${CMAKE_CURRENT_SOURCE_DIR}/${IDL}.h)
	set(IDL_INPUT ${CMAKE_CURRENT_SOURCE_DIR}/${IDL}.idl)

	add_custom_command(
		OUTPUT  ${IDL_OUT}
		DEPENDS ${IDL_INPUT}
		COMMAND ${MINKIDLC_BIN_DIR}/idlc ${IDL_INPUT} -o ${IDL_OUT}
		COMMENT "Generating IDL header: ${IDL_OUT}"
	)

	list(APPEND IDL_OUT_LIST ${IDL_OUT})
endforeach()

add_custom_target(
	${PROJECT_NAME}_MINKHEADERS ALL
	DEPENDS ${IDL_OUT_LIST}
)

# ---------------------------------------------------------------------------
# Executable
# ---------------------------------------------------------------------------

add_executable(${PROJECT_NAME}
	src/fsp_client.cpp
	src/fsp_util.cpp
)

add_dependencies(${PROJECT_NAME}
	${PROJECT_NAME}_MINKHEADERS
	listenercbo_MINKHEADERS
)

# ---------------------------------------------------------------------------
# Include paths
#
#   1. fsp_client/idl/        — IFSPApp.h, CFSPApp.h, IAppLoader.h, etc. (generated)
#   2. fsp_client/inc/        — fsp_internal.h
#   3. listenercbo/idl/       — IClientEnv.h (generated by listenercbo target)
#   4. libminkadaptor/include/ — object.h, MinkCom.h
# ---------------------------------------------------------------------------

target_include_directories(${PROJECT_NAME}
	PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/idl
	PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/inc
	PRIVATE ${CMAKE_SOURCE_DIR}/listeners/listenercbo/idl
	PRIVATE ${CMAKE_SOURCE_DIR}/libminkadaptor/include
)

# ---------------------------------------------------------------------------
# Link libraries
# ---------------------------------------------------------------------------

target_link_libraries(${PROJECT_NAME}
	PRIVATE minkadaptor
)

# ---------------------------------------------------------------------------
# Install
# ---------------------------------------------------------------------------

install(TARGETS ${PROJECT_NAME}
	DESTINATION "${CMAKE_INSTALL_BINDIR}"
)

# ---------------------------------------------------------------------------
# Host-only unit tests (pure logic, no QTEE/MinkCom calls)
# ---------------------------------------------------------------------------

if(BUILD_UNITTEST)
	add_executable(fsp_client_unittest
		src/fsp_util.cpp
		tests/unit/test_fsp_util.cpp
	)

	add_dependencies(fsp_client_unittest
		${PROJECT_NAME}_MINKHEADERS
	)

	target_include_directories(fsp_client_unittest
		PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/idl
		PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/inc
		PRIVATE ${CMAKE_SOURCE_DIR}/libminkadaptor/include
	)
endif()
